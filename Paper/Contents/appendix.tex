\section{Detailed Proof of Theorem~\ref{thm:security}}
\label{app:security-proof}

\begin{proof}
We establish completeness and soundness for the interactive protocol, and then
explain the Fiat--Shamir variant.

\paragraph{Perfect completeness}
Suppose the prover is honest and $\mat{A}\mat{B} = \mat{C}$.
Let $\mat{D} \coloneqq \mat{A}\mat{B} - \mat{C} = \mat{0}$.
For any challenge vector $\vect{r}$, we have $\vect{r}\mat{D}=\vect{0}$, so
\[
  \vect{y}
  = \vect{x}\mat{B}
  = (\vect{r}\mat{A})\mat{B}
  = \vect{r}(\mat{A}\mat{B})
  = \vect{r}\mat{C}
  = \vect{z}.
\]
For each intermediate vector $v\in\{\vect{x},\vect{y},\vect{z}\}$, the honest
prover commits to the intermediate vector before $I$ is sampled, commits to the
encoded leaf handles, publishes SNARK-side commitments through the
CC-SNARK backend,
and supplies accepted Merkle openings and CP-Link proofs for both the full
vectors and the sampled coordinates.
For any queried index $j\in[n]$, linearity of $\mathcal{C}$ gives
\[
  \fold(\enc(\mat{A})^{(j)},\vect{r})
  = \enc(\vect{r}\mat{A})_j
  = \enc(\vect{x})_j,
\]
and analogously for the $\mat{B}$ and $\mat{C}$ folds.
Since $\vect{y}=\vect{z}$, the equality check between their encoded openings
also holds at every coordinate.
All authenticated openings and backend checks are honest, so the verifier
accepts with probability one.

\paragraph{Bad-event decomposition}
Let $\mathsf{Acc}$ be the event that the verifier accepts a false committed
matrix statement, and assume $\mat{A}\mat{B}\neq\mat{C}$.
Define $\mathsf{Bad}_{\mathsf{SNARK}}$ as acceptance of a false in-circuit
SNARK instance, $\mathsf{Bad}_{\mathsf{cc}}$ as a vector-level or
witness-to-leaf linking failure, and $\mathsf{Bad}_{\mathsf{bind}}$ as two
inconsistent accepted leaf-handle openings for the same root and position.
For $\mathsf{Bad}_{\mathsf{cc}}$, condition on the CC-SNARK binding property:
for every accepted SNARK-side vector commitment $\Gamma^{\mathsf{vec}}_v$ it
obtains the vector witness $v$ and opening randomness bound to that commitment;
for every accepted sampled commitment $\Gamma_{\chi,j}$ it obtains the sampled
witness value $a_{\chi,j}$ and opening randomness $\rho_{\chi,j}$ bound to that
commitment.
The CP-Link batch tag is
$T_I=\Hash(\textsf{FMM-link-batch},\mathsf{Cm}_{ABC},\mathsf{Cm}_{XYZ},I,P_I)$,
where
\[
P_I=
\{(v,\Gamma^{\mathsf{vec}}_v,\mu_v)\}_{v\in\mathcal{W}}
\cup
\{(\chi,j,\Gamma_{\chi,j},L_{\chi,j})\}_{\chi,j}.
\]
Thus the auxiliary proof is tied to the same vector-level and sampled commitment
pairs that appear in the SNARK public statement and the Merkle openings.
A vector-level linking failure occurs if the CP-Link proof accepts but, for some
$v\in\mathcal{W}$, $\Gamma^{\mathsf{vec}}_v$ and $\mu_v$ do not open to the same
vector.
A sampled linking failure occurs if
$\merkleO(\pi_{\chi,j},L_{\chi,j},\cm_\chi,j)=1$ and
$\cpLinkV(\para,T_I,P_I,\tau_I)=1$, but
for some sampled pair there is no leaf-opening randomness $\eta_{\chi,j}$ such
that $L_{\chi,j}$ opens to the same $a_{\chi,j}$ extracted from the Groth16
witness.
For the Pedersen handle instantiation this means there is no $\eta_{\chi,j}$
with
\begin{align*}
  L_{\chi,j}
  &= \eta_{\chi,j}H+
  \sum_{\ell=1}^{d_\chi} (a_{\chi,j})_\ell G_\ell,\\
  \Gamma_{\chi,j}-L_{\chi,j}
  &= (\rho_{\chi,j}-\eta_{\chi,j})H .
\end{align*}
Merkle position binding is charged separately to
$\mathsf{Bad}_{\mathsf{bind}}$; outside that event, the accepted
$L_{\chi,j}$ is the unique handle at position $j$ under the pre-query root
$\cm_\chi$.
By the assumed backend properties,
\begin{align*}
\Pr[\mathsf{Bad}_{\mathsf{SNARK}}]
  &\le \epsilon_{\mathsf{SNARK}}(\lambda),\\
\Pr[\mathsf{Bad}_{\mathsf{cc}}]
  &\le \epsilon_{\mathsf{cc}}(\lambda),\\
\Pr[\mathsf{Bad}_{\mathsf{bind}}]
  &\le \epsilon_{\mathsf{bind}}(\lambda).
\end{align*}
Conditioned on the complement of these events, every accepting transcript has a
valid SNARK witness, every accepted intermediate vector commitment binds a
single vector $v$ before $I$ is sampled, every sampled value used by the
Groth16 fold relation is bound to a SNARK-side commitment, every accepted
auxiliary opening is linked to that commitment, and every accepted leaf handle is
uniquely determined by its root and queried coordinate.
The input roots are fixed before the Freivalds challenge is sampled,
and the intermediate commitments are fixed before the sampled query tuple is
sampled.

\paragraph{Freivalds randomization}
Under the same conditioning, the matrices bound to
$\cm_A,\cm_B,\cm_C$ are fixed and satisfy
$\mat{A}\mat{B}\neq\mat{C}$.
Let $\mat{D}=\mat{A}\mat{B}-\mat{C}\neq 0$.
There exists a column $\vect{d}_j$ of $\mat{D}$ with $\vect{d}_j\neq 0$.
The condition $\vect{r}\mat{D}=0$ implies
\[
  P_j(r) \coloneqq
  \langle \vect{r},\vect{d}_j\rangle
  = \sum_{i=0}^{k-1}(\mat{D})_{i,j} r^i
  = 0 .
\]
Since $P_j$ is a nonzero polynomial of degree at most $k-1$ and $r$ is sampled
after the input commitments are fixed, Lemma~\ref{lem:sz} gives
\[
  \Pr_{r\sample\mathbb{F}}[\vect{r}\mat{D}=0]
  \le \Pr_{r\sample\mathbb{F}}[P_j(r)=0]
  \le \frac{k-1}{|\mathbb{F}|}.
\]

\paragraph{Proximity sampling}
Now condition further on $\vect{r}\mat{D}\neq 0$.
No vectors can simultaneously satisfy all four global relations:
\[
  \vect{x}=\vect{r}\mat{A},\quad
  \vect{y}=\vect{x}\mat{B},\quad
  \vect{z}=\vect{r}\mat{C},\quad
  \vect{y}=\vect{z}.
\]
Otherwise the four equalities would imply
$\vect{r}\mat{A}\mat{B}=\vect{r}\mat{C}$, contradicting
$\vect{r}\mat{D}\neq 0$.
Thus at least one relation is false.
Consider $\vect{x}\neq\vect{r}\mat{A}$; the other cases are identical.
Injectivity and distance of the code imply that the two codewords
$\enc(\vect{x})$ and $\enc(\vect{r}\mat{A})$ disagree on at least
$\delta n$ coordinates.
By code linearity, the sampled fold value from the committed rows of
$\mat{A}$ equals $\enc(\vect{r}\mat{A})_j$ at coordinate $j$.
Since the sampled column used by Groth16 is bound to the authenticated input
leaf handle, and the sampled vector value is both bound to its authenticated
leaf handle and checked to equal the circuit-computed codeword
$\enc(\vect{x})_j$, the fold check rejects whenever $j$ is in the disagreement
set.
A single uniformly sampled coordinate misses this disagreement set with
probability at most $1-\delta$, and $t$ independent sampled coordinates all
miss it with probability at most $(1-\delta)^t$.
Union-bounding over the four possible violated relations yields
$4(1-\delta)^t$.

\paragraph{Combining the bounds and parameters}
Summing the backend, Freivalds, and proximity errors gives
\[
\begin{aligned}
\Pr[\mathsf{Acc}\mid\mat{A}\mat{B}\neq\mat{C}]
&\le
\epsilon_{\mathsf{SNARK}}(\lambda)
+\epsilon_{\mathsf{cc}}(\lambda)
+\epsilon_{\mathsf{bind}}(\lambda)\\
&\quad+\frac{k-1}{|\mathbb{F}|}+4(1-\delta)^t .
\end{aligned}
\]
For $\lambda=128$, a 256-bit field makes $(k-1)/|\mathbb{F}|$ negligible for
the reported dimensions.
The proximity term is at most $2^{-\lambda}$ when
$t=\lceil(\lambda+2)/\log_2(1/(1-\delta))\rceil$; for example, for a code with
$\delta=1/2$, $t=130$ suffices for the proximity term alone.
For a conservative aggregate 128-bit budget over all terms, one should allocate
the budget across terms, e.g., by targeting each term at $2^{-131}$.

\paragraph{Fiat--Shamir variant}
The non-interactive transcript derives both $r$ and $I$ from
domain-separated hashes of the transcript prefix, including the public input
roots, intermediate vector commitments, and encoded-vector roots.
In the random oracle model, the preceding public-coin proof applies with the
standard Fiat--Shamir loss $\epsilon_{\mathsf{FS}}(q_H)$ for an adversary that
makes $q_H$ random-oracle queries.
This proves Theorem~\ref{thm:security}.
\end{proof}

\section{CP-Link Backend Instantiation}
\label{app:cplink-backend}

The artifact uses Pedersen-style~\cite{pedersen1991non} commitments
\[
  \Gamma=\rho_\Gamma H+\sum_{\ell=1}^{d_\chi} a_\ell G_\ell ,
  \qquad
  C=\rho_C H+\sum_{\ell=1}^{d_\chi} a_\ell G_\ell ,
\]
with $d_\chi=k$ for matrix columns and full intermediate vectors, and
$d_\chi=1$ for vector coordinates.
For vector-level links, $C$ is the pre-query vector commitment $\mu_v$.
For sampled links, $C$ is the authenticated Merkle leaf handle $L_{\chi,j}$.
Leaf handles are authenticated by the Merkle tree, while the CC-SNARK backend
binds $\Gamma$ to the vector or sampled witness value used by the circuit.
The link proof shows that the SNARK-side commitment and the external commitment
open to the same object.

\begin{theorem}[Batched CP-Link backend]\label{thm:cp-link-backend}
Assume the Pedersen handle parameters are generated so that the handle
commitment is binding, the QA-NIZK link proof for the linear batch relation is
sound, and the batching challenge is sampled by the Fiat--Shamir random oracle
after the batch statement is fixed.
For $Q=|\mathcal{S}|t+|\mathcal{W}|$ vector-level and sampled objects, the
batched CP-Link verifier accepts a false batch with probability at most
$\epsilon_{\mathsf{ped}}+\epsilon_{\mathsf{QA}}+Q/|\mathbb{F}|$.
The artifact serializes the $Q$ logical relations as one online batched proof;
generator and CRS bytes are setup material and are not included in the online
proof-size claim.
\end{theorem}

The logical CP-Link relation contains one equation per vector-level or sampled
object:
\[
  \Gamma_q-C_q=\Delta_qH,\qquad q\in[Q],
\]
where $Q=|\mathcal{S}|t+|\mathcal{W}|$ and $q$ indexes both vector-level pairs
$(v,\Gamma^{\mathsf{vec}}_v,\mu_v)$ and sampled pairs
$(\chi,j,\Gamma_{\chi,j},L_{\chi,j})$.
The batch statement is
\[
\begin{aligned}
P_I&=\{(\mathrm{tag}_q,\Gamma_q,C_q)\}_{q=1}^{Q},\\
T_I&=\Hash(\textsf{FMM-link-batch},\mathsf{Cm}_{ABC},\mathsf{Cm}_{XYZ},I,P_I).
\end{aligned}
\]
The implementation batches these equations by deriving
$\beta\gets\Hash(\textsf{FMM-link-rand},T_I,\{\Gamma_q,C_q\}_{q=1}^{Q})$ and
checking a single randomized linear relation
\[
  \sum_{q=1}^{Q}\beta^q(\Gamma_q-C_q)=
  \left(\sum_{q=1}^{Q}\beta^q\Delta_q\right)H .
\]
If any individual equality is false, the left-minus-right side is a nonzero
degree-$Q$ polynomial in $\beta$, so the batch can pass with probability at
most $Q/|\mathbb{F}|$ in the random-oracle model.
Combining this with Pedersen binding and QA-NIZK soundness gives the bound in
Theorem~\ref{thm:cp-link-backend}.
The QA-NIZK statement is
$(\para,T_I,\{\Gamma_q,C_q\}_{q=1}^{Q})$ and the witness is the aggregate
difference opening $\sum_q\beta^q\Delta_q$.
Its CRS fixes the handle generators and linear-relation verification
parameters; if serialized, those bytes are setup/key/CRS material rather than
online proof bytes.
Thus the $128$-byte value reported in the experiments is one online batched
QA-NIZK proof for all vector-level and sampled links, not a per-link cost.
